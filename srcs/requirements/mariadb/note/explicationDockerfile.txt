FROM debian:bookworm = on part d'une image de base de Debian version bookworm
le container demarre avec un mini systeme debian, rien d'autre n'est installer
on choisit debian car on utilise des bases de donnees complexe donc debian plus sur
le choix d'alpine ou debian depend du logiciel et de ses besoins internes

RUN apt update && apt upgrade -y && \
	apt install mariadb-server -y && \
	rm -rf /var/lib/apt/list/*
instruction executer au moment du build, pas au runtime
apt update = maj la liste des paquets disponibles
apt upgrade -y = maj les paquets existants
apt install mariadb-server -y = installe le serveur mariadb
rm -rf /var/lib/apt/list/* = supprime les cache apt pour reduire la taille de l'image et eviter d'avoir des fichiers inutiles

COPY conf/my.cnf /etc/mysql/my.conf = copie le fichier de configuration personnaliser dans le container
cad on remplace la config par defaut
ici bind-address=0.0.0.0 qui permet aux autres containers (wordpress) de se connecter
bind-address determine sur quelle interface reseau MariaDB ecoute
127.0.0.1 = ecoute seulement en local (localhost)
0.0.0.0 = ecoute sur tte les interfaces reseau
car dans inception mariadb tourne dans un conteneur et wordpress dans un autre et les 2 communiquent via le reseau docker
si on laisse bind-address = 127.0.0.1 alors
mariadb acceptera uniquement les connexions venant de lui meme (dans son propre conteneur)
et le conteneur wordpress ne pourra pas se connecter, il y aura une erreur du type "can't connect to database"
si on met bind-address = 0.0.0.0 alors mariadb accepte les connexions venant du reseau docker et des autres conteneurs

COPY tools/init.sh /init.sh = copie le script d'initialisation dans le container
ce script va creer la base, creer l'utilisateur, configurer les permissions et demarrer mariadb

RUN chmod +x /init.sh = donne les droits d'execution au script, sinon docker ne pourrait pas le lancer

EXPOSE 3306 = indique que le container utilise le port 3306 (port mysql/mariadb)
ca ouvre pas le port, ca documente juste l'intention
c'est docker-compose qui fera la vraie connexion reseau

CMD["/init.sh"] = commande executer quand le container demarre
docker va lancer /init.sh

RUN = pendant le build
CMD = au lancement du container
COPY = pendant le build
EXPOSE = documentation
